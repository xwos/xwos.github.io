230 xwer_t wait(enum Trigger trigger, bool consumption,
231 xwos::Bmp<TNum> * origin, xwos::Bmp<TNum> * msk,
232 enum WaitMode mode = WaitMode::FlgWait, xwtm_t to = XWTM_MAX) {
233 xwer_t rc;
234 switch (mode) {
235 case WaitMode::FlgWait:
236 rc = xwos_flg_wait(mFlgPtr, (enum xwos_flg_trigger_em)trigger,
237 consumption, origin->mData, msk->mData);
238 break;
239 case WaitMode::FlgWaitTimed:
240 rc = xwos_flg_wait_to(mFlgPtr, (enum xwos_flg_trigger_em)trigger,
241 consumption, origin->mData, msk->mData, to);
242 break;
243 case WaitMode::FlgWaitUninterruptable:
244 rc = xwos_flg_wait_unintr(mFlgPtr, (enum xwos_flg_trigger_em)trigger,
245 consumption, origin->mData, msk->mData);
246 break;
247 case WaitMode::FlgTryWait:
248 rc = xwos_flg_trywait(mFlgPtr, (enum xwos_flg_trigger_em)trigger,
249 consumption, origin->mData, msk->mData);
250 break;
251 default:
252 rc = xwos_flg_wait(mFlgPtr, (enum xwos_flg_trigger_em)trigger,
253 consumption, origin->mData, msk->mData);
254 break;
255 }
256 return rc;
